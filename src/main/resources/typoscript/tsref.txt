# TypoScript reference data for T3 Toolkit (own compilation, Apache-2.0).
#
# @type NAME [: PARENT]     a type; PARENT's properties are inherited
# @root setup|tsconfig      top-level objects
# name: TYPE | text         a property; TYPE "string/stdWrap" = value with stdWrap properties
#                           "#" as name = numeric keys (10, 20 …), "*" = any key
# name: TYPE = a,b,c | text known values

@root setup
page: PAGE | The page object - configure what the frontend renders.
config: CONFIG | Global configuration of the page rendering.
lib: LIB | Library objects to reference or copy elsewhere.
plugin: PLUGINS | Configuration of plugins and Extbase extensions (plugin.tx_*).
module: PLUGINS | Configuration of backend modules (module.tx_*).
tt_content: CASE | Rendering of content elements by CType.
styles: LIB | Objects of static templates.
temp: LIB | Temporary objects, removed after parsing.
sitetitle: string | Website title.
types: ANY | Registered type numbers.

@root tsconfig
